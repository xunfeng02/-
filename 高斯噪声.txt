class ReverseDPTopK:
    def __init__(self, epsilon=1.0, delta=1e-5):
        self.epsilon = epsilon
        self.delta = delta  # 高斯噪声必须参数

    # 高斯噪声（差分隐私标准公式）
    def gaussian_noise(self, sens=1.0):
        sigma = np.sqrt(2 * np.log(1.25 / self.delta)) * sens / self.epsilon
        return np.random.normal(loc=0, scale=sigma)

    def run_with_noisy_scores(self, item_counts: Counter, k=5):
        items = list(item_counts.keys())
        counts = list(item_counts.values())

        sorted_counts = sorted(counts, reverse=True)
        tau = sorted_counts[k-1] if len(counts) >= k else sorted_counts[-1]

        noisy_scores = []
        noises = []
        for c in counts:
            noise = self.gaussian_noise()
            noisy_score = c + noise
            noises.append(noise)
            noisy_scores.append(noisy_score)

        combined = list(zip(items, counts, noisy_scores))
        combined_sorted = sorted(combined, key=lambda x: x[2], reverse=True)
        topk_result = combined_sorted[:k]

        return topk_result, noises, noisy_scores
