import numpy as np
import pandas as pd
from collections import Counter
from scipy.stats import kendalltau
import warnings
warnings.filterwarnings('ignore')

cols = [
    "age", "workclass", "fnlwgt", "education", "education_num",
    "marital_status", "occupation", "relationship", "race", "sex",
    "capital_gain", "capital_loss", "hours_per_week", "native_country", "income"
]

df = pd.read_csv("census.csv", names=cols, skipinitialspace=True, na_values="?")
feature = "occupation" 
K = 5
delta = 1e-5
sens = 1.0
c_total = len(df)

# 数据
data = df[feature].dropna()
true_counter = Counter(data)
true_list = sorted(true_counter.items(), key=lambda x:-x[1])
true_topk_items = [i for i,c in true_list[:K]]
true_topk_counts = [c for i,c in true_list[:K]]

# 评价指标
def hit_rate(pred, true):
    return len(set(pred) & set(true)) / K

def mean_relative_error(noisy_counts, true_counts):
    return np.mean(np.abs(np.array(noisy_counts)-np.array(true_counts))/(np.array(true_counts)+1e-8))

def kendall_score(pred_items, true_items):
    rank_pred = {item:i for i,item in enumerate(pred_items)}
    rank_true = {item:i for i,item in enumerate(true_items)}
    pred_rank = [rank_pred[i] for i in true_items if i in rank_pred]
    true_rank = list(range(len(pred_rank)))
    return kendalltau(pred_rank, true_rank)[0]

# RDP对齐核心
from scipy.optimize import minimize_scalar
def compute_pure_dp_to_rdp(eps_target, delta):
    def rdp_loss(alpha):
        eps_rdp = eps_target + (np.log(1/delta)+(alpha/2)*np.log(2*np.pi*alpha))/(alpha-1)
        return eps_rdp
    res = minimize_scalar(rdp_loss, bounds=(2,100), method='bounded')
    return res.fun

# 三种机制
def laplace_rdp_aligned(eps_target, delta):
    eps_rdp = compute_pure_dp_to_rdp(eps_target, delta)
    scale = sens/eps_rdp
    items = list(true_counter.keys())
    counts = np.array(list(true_counter.values()))
    noisy = counts + np.random.laplace(0, scale, len(counts))
    ranked = sorted(zip(items, counts, noisy), key=lambda x:-x[2])
    pred_items = [i for i,c,n in ranked[:K]]
    pred_counts = [c for i,c,n in ranked[:K]]
    return pred_items, pred_counts

def standard_gaussian_aligned(eps_target, delta):
    sigma = np.sqrt(2*np.log(1.25/delta))*sens/eps_target
    items = list(true_counter.keys())
    counts = np.array(list(true_counter.values()))
    noisy = counts + np.random.normal(0, sigma, len(counts))
    ranked = sorted(zip(items, counts, noisy), key=lambda x:-x[2])
    pred_items = [i for i,c,n in ranked[:K]]
    pred_counts = [c for i,c,n in ranked[:K]]
    return pred_items, pred_counts

def tight_pure_gaussian_rdp_aligned(eps_target, delta, c):
    eps_rdp = compute_pure_dp_to_rdp(eps_target, delta)
    sigma = np.sqrt(2*(c+sens))/eps_rdp
    items = list(true_counter.keys())
    counts = np.array(list(true_counter.values()))
    noisy = counts + np.random.normal(0, sigma, len(counts))
    ranked = sorted(zip(items, counts, noisy), key=lambda x:-x[2])
    pred_items = [i for i,c,n in ranked[:K]]
    pred_counts = [c for i,c,n in ranked[:K]]
    return pred_items, pred_counts

# 主实验
eps_list = [0.1, 0.5, 1.0, 2.0, 5.0]
print("="*90)
print(" RDP隐私对齐后公平对比（增加Kendall+MRE敏感指标）")
print("="*90)

for eps in eps_list:
    print(f"\n 目标隐私 ε = {eps}")
    # Laplace
    pred_lap_items, pred_lap_cnt = laplace_rdp_aligned(eps, delta)
    hit_lap = hit_rate(pred_lap_items, true_topk_items)
    mre_lap = mean_relative_error(pred_lap_cnt, true_topk_counts)
    tau_lap = kendall_score(pred_lap_items, true_topk_items)
    # Standard Gaussian
    pred_gau_items, pred_gau_cnt = standard_gaussian_aligned(eps, delta)
    hit_gau = hit_rate(pred_gau_items, true_topk_items)
    mre_gau = mean_relative_error(pred_gau_cnt, true_topk_counts)
    tau_gau = kendall_score(pred_gau_items, true_topk_items)
    # Tight Pure DP Gaussian
    pred_tight_items, pred_tight_cnt = tight_pure_gaussian_rdp_aligned(eps, delta, c_total)
    hit_tight = hit_rate(pred_tight_items, true_topk_items)
    mre_tight = mean_relative_error(pred_tight_cnt, true_topk_counts)
    tau_tight = kendall_score(pred_tight_items, true_topk_items)

    print(f"{'方法':<22}{'Hit Rate':<10}{'Kendall-τ':<12}{'MRE':<10}")
    print(f"Laplace(纯DP→RDP)   {hit_lap:.2f}      {tau_lap:.3f}      {mre_lap:.3f}")
    print(f"Standard Gaussian   {hit_gau:.2f}      {tau_gau:.3f}      {mre_gau:.3f}")
    print(f"Tight Pure Gaussian {hit_tight:.2f}      {tau_tight:.3f}      {mre_tight:.3f}")
